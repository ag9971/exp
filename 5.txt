hasLUBs[R_?partialorderQ] := Module[{domR, a, b}, domR = findDomain[R];
  Catch[Do[If[leastUpperBound[R, {a, b}] === "Does not exists", Throw[False]], {a, domR}, {b, domR}];
   Throw[True]]]
hasGLBs[R_?partialorderQ] := Module[{domR, a, b}, domR = findDomain[R];
  Catch[Do[If[greatestLowerBound[R, {a, b}] === "Does not exists", Throw[False]], {a, domR}, {b, domR}];
   Throw[True]]]
hasGLB1s[R_?partialorderQ] := hasLUBs[inverseRelation[R]]
latticeQ[R_?partialorderQ] := hasLUBs[R] && hasGLBs[R]
